    static void Pump(int ms = 80)
    {
        var frame = new DispatcherFrame(); var timer = new DispatcherTimer { Interval = TimeSpan.FromMilliseconds(ms) };
        timer.Tick += (_, _) => { timer.Stop(); frame.Continue = false; }; timer.Start(); Dispatcher.PushFrame(frame);
    }
    static void Until(Func<bool> condition)
    { for (int i = 0; i < 80 && !condition(); i++) Pump(50); Assert(condition(), "Asynchronous form operation completed"); }
    static object? Call(object target, string method, params object?[] args) => target.GetType().GetMethod(method, BindingFlags.NonPublic | BindingFlags.Instance)!.Invoke(target, args);
    static T Field<T>(UserControl form, string name) => (T)form.FindName(name);
    static void Theme(bool dark)
    {
        string code = System.IO.File.ReadAllText("MainWindow.xaml.cs");
        int start = code.IndexOf("private void SetTheme(bool isDark)");
        int first = code.IndexOf("res[\"AppWindowBg\"]", start), second = code.IndexOf("res[\"AppWindowBg\"]", first+1);
        string section = dark ? code[first..second] : code[second..];
        foreach (Match m in Regex.Matches(section,"res\\[\"([^\"]+)\"\\]\\s*=\\s*CreateBrush\\(\"([^\"]+)\""))
            Application.Current.Resources[m.Groups[1].Value] = new SolidColorBrush((Color)ColorConverter.ConvertFromString(m.Groups[2].Value));
        foreach (Match m in Regex.Matches(section,"res\\[\"([^\"]+)\"\\]\\s*=\\s*CreateGradient\\(\"([^\"]+)\",\\s*\"([^\"]+)\""))
            Application.Current.Resources[m.Groups[1].Value] = new LinearGradientBrush((Color)ColorConverter.ConvertFromString(m.Groups[2].Value),(Color)ColorConverter.ConvertFromString(m.Groups[3].Value),45);
    }
    static Window Host(UIElement content, double width=1300, double height=900)
    {
        var root = new Grid { Background = Application.Current.TryFindResource("AppWindowBg") as Brush ?? Brushes.White };
        root.Children.Add(new AdornerDecorator { Child = content, Margin = new Thickness(18) });
        var w = new Window { Content=root, Width=width, Height=height, Left=-12000, Top=-12000, ShowInTaskbar=false, ShowActivated=false };
        w.Show(); Pump(140); return w;
    }
    static void Capture(Window w, string name)
    {
        foreach (var close in Find<Button>(w).Where(b=>b.Content?.ToString()=="×").ToArray()) close.RaiseEvent(new RoutedEventArgs(Button.ClickEvent));
        Pump(280); var root=(FrameworkElement)w.Content;
        var bitmap=new RenderTargetBitmap((int)root.ActualWidth,(int)root.ActualHeight,96,96,PixelFormats.Pbgra32); bitmap.Render(root);
        var png=new PngBitmapEncoder();png.Frames.Add(BitmapFrame.Create(bitmap));
        using var stream=System.IO.File.Create(System.IO.Path.Combine(output,name+".png"));png.Save(stream);
    }
    static IEnumerable<T> Find<T>(DependencyObject root) where T:DependencyObject
    {
        if(root is T found)yield return found;
        for(int i=0;i<VisualTreeHelper.GetChildrenCount(root);i++)foreach(var c in Find<T>(VisualTreeHelper.GetChild(root,i)))yield return c;
    }
    static void Dentistry()
    {
        Assert(ToothGlyph.CrownGeometry(11).ToString()!=ToothGlyph.CrownGeometry(12).ToString(),"Central and lateral incisors differ");
        Assert(ToothGlyph.CrownGeometry(11).ToString()!=ToothGlyph.CrownGeometry(31).ToString(),"Upper and lower incisors differ");
        Assert(ToothGlyph.RootGeometry(16).ToString()!=ToothGlyph.RootGeometry(46).ToString(),"Upper and lower molar roots differ");
        Assert(ToothGlyph.RootGeometry(14).ToString()!=ToothGlyph.RootGeometry(15).ToString(),"First and second upper premolar roots differ");
        Assert(ToothGlyph.CrownGeometry(54).ToString()!=ToothGlyph.CrownGeometry(14).ToString(),"Primary molars are not drawn as premolars");
        Assert(ToothGlyph.RootGeometry(55).ToString()!=ToothGlyph.RootGeometry(16).ToString(),"Primary molars have different root proportions");
        foreach(var type in new[]{typeof(AdultDentalChart),typeof(ChildrenDentalChart),typeof(MyClinic.OriginalTheme.AdultDentalChart),typeof(MyClinic.OriginalTheme.ChildrenDentalChart)})
        {
            var chart=(UserControl)Activator.CreateInstance(type)!;var window=Host(chart,1050,520);
            var buttons=Find<ToggleButton>(chart).ToList();bool child=type.Name.Contains("Children");
            Assert(buttons.Count==(child?20:32),"Chart has every FDI tooth exactly once");
            Assert(buttons.Select(b=>b.Content.ToString()).Distinct().Count()==buttons.Count,"FDI identities are unique");
            Assert(buttons.All(b=>ToothGlyph.IsValidId(int.Parse(b.Content.ToString()!))),"All FDI identities are valid");
            foreach(var b in buttons){b.IsChecked=true;}
            Assert(((System.Collections.IEnumerable)type.GetMethod("GetSelectedTeeth")!.Invoke(chart,null)!).Cast<string>().Count()==buttons.Count,"Every chart tooth can be selected");
            type.GetMethod("ClearSelection")!.Invoke(chart,null);Assert(buttons.All(b=>b.IsChecked==false),"Clear selection resets every tooth");
            buttons[0].IsChecked=true;Capture(window,(type.Namespace!.Contains("Original")?"original":"modern")+(child?"-primary-chart":"-adult-chart"));window.Close();
        }
        CultureInfo previous=CultureInfo.CurrentCulture;
        try { CultureInfo.CurrentCulture=new CultureInfo("fr-FR"); for(int q=1;q<=8;q++)for(int n=1;n<=(q<=4?8:5);n++){ToothGlyph.CrownGeometry(q*10+n);ToothGlyph.RootGeometry(q*10+n);} }
        finally { CultureInfo.CurrentCulture=previous; }
        Assert(true,"All geometries parse in decimal-comma locales");
    }
    static void UpgradeAndRecovery()
    {
        string databasePath=System.IO.Path.GetFullPath("obj/ClinicFormSmoke/upgrade-"+Guid.NewGuid().ToString("N")+".db");
        var options=new DbContextOptionsBuilder<AppDbContext>().UseSqlite("Data Source="+databasePath).Options;
        using(var db=new AppDbContext(options))db.Database.ExecuteSqlRaw("CREATE TABLE Patients (Id INTEGER PRIMARY KEY, FullName TEXT); INSERT INTO Patients VALUES (1, 'Synthetic existing patient');");
        for(int i=0;i<2;i++)using(var db=new AppDbContext(options))typeof(MyClinic.App).GetMethod("EnsureColumnExists",BindingFlags.NonPublic|BindingFlags.Static)!.Invoke(null,new object[]{db,"Patients","DateOfBirth","TEXT NULL"});
        for(int i=0;i<2;i++)using(var db=new AppDbContext(options))typeof(MyClinic.App).GetMethod("EnsureColumnExists",BindingFlags.NonPublic|BindingFlags.Static)!.Invoke(null,new object[]{db,"Patients","BirthYear","INTEGER NULL"});
        using(var connection=new SqliteConnection("Data Source="+databasePath))
        {
            connection.Open();using var command=connection.CreateCommand();command.CommandText="SELECT FullName FROM Patients WHERE Id=1 AND DateOfBirth IS NULL AND BirthYear IS NULL";
            Assert((string?)command.ExecuteScalar()=="Synthetic existing patient","DOB upgrade is idempotent and preserves existing patient data");
        }
        using var memory=new SqliteConnection("Data Source=:memory:");memory.Open();
        var memoryOptions=new DbContextOptionsBuilder<AppDbContext>().UseSqlite(memory).Options;
        AppDbContext Context()=>new(memoryOptions);using(var db=Context())db.Database.EnsureCreated();
        string draftPath=System.IO.Path.GetFullPath("obj/ClinicFormSmoke/recovery.draft");if(System.IO.File.Exists(draftPath))System.IO.File.Delete(draftPath);
        var form=new AddPatientView(Context,false,draftPath);var window=Host(form);Until(()=>Field<TextBox>(form,"TxtFullName").IsEnabled);
        Field<TextBox>(form,"TxtFullName").Text="Recoverable synthetic patient";
        Field<TextBox>(form,"TxtDiagnosis").Text="Unfinished clinical note";
        Assert(form.PreserveDraft()&&System.IO.File.Exists(draftPath),"Leaving an edited form writes an encrypted draft");window.Close();
        var reopened=new AddPatientView(Context,false,draftPath);window=Host(reopened);Until(()=>Field<TextBox>(reopened,"TxtFullName").IsEnabled);
        Assert(Field<TextBox>(reopened,"TxtFullName").Text=="Recoverable synthetic patient"&&Field<TextBox>(reopened,"TxtDiagnosis").Text=="Unfinished clinical note","Reopening recovers unsaved identity and clinical notes");
        Field<TextBox>(reopened,"TxtFullName").Clear();Field<TextBox>(reopened,"TxtDiagnosis").Clear();
        Assert(reopened.PreserveDraft()&&!System.IO.File.Exists(draftPath),"Reverting to the empty baseline removes obsolete recovered draft data");window.Close();
        var blocked=new DraftGuard();var host=new ContentControl{Content=blocked};
        Assert(!NavigationTransition.Show(host,new Border())&&ReferenceEquals(host.Content,blocked),"Failed draft preservation keeps the current form visible");
        blocked.Allow=true;Assert(NavigationTransition.Show(host,new Border())&&host.Content is Border,"Navigation proceeds after draft preservation succeeds");
    }
    sealed class DraftGuard:UserControl,IRecoverableClinicForm { public bool Allow; public bool PreserveDraft()=>Allow; }
    static void Form(Type type, bool dark)

        using(var db=Context())
        {

            db.TreatmentCosts.Add(new TreatmentCost {TreatmentName="تنظيف تجريبي",Cost=50,Currency="SYP"});db.SaveChanges();

        Theme(dark);

        Assert(errorLabel!=null&&errorLabel.FlowDirection==FlowDirection.RightToLeft,"Arabic validation uses native RTL text rather than a mirrored drawing");
        Pump();

        Field<TextBox>(form,"TxtBirthYear").Text=(AppClock.Today.Year-30).ToString(CultureInfo.InvariantCulture);
        Assert(Field<TextBox>(form,"TxtAge").Text=="30" && Field<TextBox>(form,"TxtAge").IsReadOnly,"Year-only input calculates approximate age without asking for a month or day");

        Assert(Field<TextBlock>(form,"TxtIdentityAlert").Text.Contains("حساسية تجريبية"),"Persistent identity strip shows documented allergy");

        var selected=(System.Collections.ObjectModel.ObservableCollection<TreatmentSelectionItem>)type.GetField("_treatmentSelectionItems",BindingFlags.NonPublic|BindingFlags.Instance)!.GetValue(form)!;

        Field<TextBox>(form,"TxtCurrentCost").Text="١٠٠";Field<TextBox>(form,"TxtTodayPaid").Text="١٥٠";
        Call(form,"BtnSaveVisit_Click",Field<Button>(form,"BtnSaveVisit"),new RoutedEventArgs());Pump();

    static double Contrast(Brush foreground,Brush background)
    {
        static double L(Color color)
        {
            static double Channel(byte channel){double v=channel/255.0;return v<=.04045?v/12.92:Math.Pow((v+.055)/1.055,2.4);}
            return .2126*Channel(color.R)+.7152*Channel(color.G)+.0722*Channel(color.B);
        }
        double f=L(((SolidColorBrush)foreground).Color),b=L(((SolidColorBrush)background).Color);
        return (Math.Max(f,b)+.05)/(Math.Min(f,b)+.05);
    }

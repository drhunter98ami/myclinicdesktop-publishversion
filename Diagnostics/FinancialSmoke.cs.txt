    static void Ledger()
    {
        Assert(FinanceMoney.TryParse("١٬٢٣٤٫٥٠",out var arabic)&&arabic==1234.50m,"Arabic digits and separators");
        Assert(FinanceMoney.TryParse("۱۲۳۴.۵۰",out var persian)&&persian==1234.50m,"Persian digits");
        Assert(!FinanceMoney.TryParse("1.234",out _)&&!FinanceMoney.TryParse("NaN",out _),"Invalid precision and non-finite values rejected");
        var rows=new[]{Visit(1,-6,150000,40000,Treatments(("تنظيف",100000,"SYP"),("حشوة",50000,"SYP"))),Visit(2,-5,-20000,0),Visit(3,-4,0,60000)};
        var account=FinancialLedger.BuildAccount(patient,rows);
        Assert(account.Cost==150000 && account.Forgiven==20000 && account.Paid==100000 && account.Balance==30000,"Discount and standalone forgiveness preserve balance");
        var payment=account.Rows.Single(r=>r.Id==3);
        Assert(payment.Treatments.Sum(t=>t.Paid)==60000 && payment.Treatments.Sum(t=>t.Remaining)==30000,"FIFO respects prior forgiveness and exact payment amount");
        Assert(account.Rows.Count(r=>!r.FinancialOnly)==1,"Payment and forgiveness entries are not treatment visits");
        var foreign=FinancialLedger.BuildAccount(patient,new[]{new Visit{Id=1,PatientId=1,VisitDate=AppClock.Today,CurrentCost=200000,TodayPaid=0,UsdToSypRateSnapshot=10000,SelectedTreatmentsJson=Treatments(("تاج",20,"USD"))}});
        Assert(foreign.Cost==200000,"Historic USD exchange rate preserved");
        var credit=FinancialLedger.BuildAccount(patient,new[]{Visit(1,-3,0,200000),Visit(2,-2,150000,0,Treatments(("علاج",150000,"SYP")))});
        Assert(credit.Balance==-50000 && credit.Rows.First().Treatments.Sum(t=>t.CreditApplied)==150000 && credit.Rows.First().Treatments.Sum(t=>t.Paid)==0,"Advance credit is distinct from cash collected in a later visit");
        var sameTime=FinancialLedger.BuildAccount(patient,new[]{Visit(2,-3,0,50000),Visit(1,-3,100000,0,Treatments(("علاج",100000,"SYP")))});
        Assert(sameTime.Rows.Single(r=>r.Id==2).Treatments.Sum(t=>t.Paid)==50000,"Same-time entries use ID order");
        Assert(FinanceDateRange.For(FinancePeriod.Month,new(2024,2,20)).End==new DateTime(2024,2,29),"Leap month period");
        Assert(FinanceDateRange.For(FinancePeriod.Custom,new(2026,1,2),new(2026,1,7)).Previous==new FinanceDateRange(new(2025,12,27),new(2026,1,1)),"Custom comparison range");
    }

    static void IncomeScrolling()
    {
        Theme(false,false);MotionPreferences.Set(MotionLevel.Off,false);
        var view=Fixture();
        var treatments=Enumerable.Range(1,40).Select(i=>("علاج "+i,2500m,"SYP")).ToArray();
        Set(view,"_visits",new List<Visit>{Visit(500,0,100000,50000,Treatments(treatments)),Visit(501,-1,10000,10000)});
        Call(view,"RebuildLedger");Call(view,"ApplyFilters");
        var window=new Window{Content=view,Width=980,Height=720,Left=-12000,Top=-12000,ShowActivated=false,ShowInTaskbar=false};window.Show();Pump();
        try
        {
            var table=Named<DataGrid>(view,"IncomeItemsControl");
            var row=(DataGridRow)table.ItemContainerGenerator.ContainerFromIndex(0);
            var toggle=Descendants<Button>(row).Single(b=>b.Content as string=="⌄ عرض");
            toggle.RaiseEvent(new RoutedEventArgs(Button.ClickEvent));Pump();
            var details=Descendants<DataGridDetailsPresenter>(row).Single();
            var scroll=Descendants<ScrollViewer>(table).First();
            Assert(details.ActualHeight>scroll.ViewportHeight,"Expanded income details exceed the viewport");
            double prior=scroll.VerticalOffset;
            for(int i=0;i<4;i++)
            {
                details.RaiseEvent(new MouseWheelEventArgs(Mouse.PrimaryDevice,Environment.TickCount,-120){RoutedEvent=Mouse.MouseWheelEvent});Pump();
                double offset=scroll.VerticalOffset;
                Assert(offset>prior && offset-prior<scroll.ViewportHeight,"Wheel advances through tall income details in small pixel steps");prior=offset;
            }
            details.RaiseEvent(new MouseWheelEventArgs(Mouse.PrimaryDevice,Environment.TickCount,120){RoutedEvent=Mouse.MouseWheelEvent});Pump();
            Assert(scroll.VerticalOffset<prior,"Wheel scrolls back inside expanded income details");
            Capture(window,"income-expanded-pixel-scroll");
            toggle.RaiseEvent(new RoutedEventArgs(Button.ClickEvent));Pump();
            Assert(row.DetailsVisibility==Visibility.Collapsed,"Income details collapse after wheel scrolling");
        }
        finally{window.Close();}
    }
